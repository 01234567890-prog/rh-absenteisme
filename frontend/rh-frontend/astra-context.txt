import React from 'react';
import { Box, Button, Card, CardContent, Chip, LinearProgress, Skeleton, Typography } from '@mui/material';
import { ArrowUpRight, Brain, CalendarDays } from 'lucide-react';
import type {
  DepartementResponseDto,
  PredictionResponseDto,
  UserResponseDto,
} from '../../types/api';
import './AbsenteeismPredictionPanel.css';

interface AbsenteeismPredictionPanelProps {
  predictions: PredictionResponseDto[];
  users: UserResponseDto[];
  departements: DepartementResponseDto[];
  loading: boolean;
  error?: string;
  onRetry: () => void;
  onViewAll?: () => void;
}

const formatScore = (score: number) => {
  if (!Number.isFinite(score)) return 'Unavailable';
  const percentage = score <= 1 ? score * 100 : score;
  return `${Math.round(percentage)}%`;
};

const formatDate = (value: string) => {
  const date = new Date(value);
  return Number.isNaN(date.getTime()) ? value : date.toLocaleDateString();
};

export const AbsenteeismPredictionPanel: React.FC<AbsenteeismPredictionPanelProps> = ({
  predictions,
  users,
  departements,
  loading,
  error,
  onRetry,
  onViewAll,
}) => {
  const userMap = new Map(users.map((user) => [user.id, user]));
  const departmentMap = new Map(departements.map((department) => [department.id, department.name]));
  const sortedPredictions = [...predictions].sort(
    (first, second) => (Number(second.score) || 0) - (Number(first.score) || 0)
  );
  const counts = predictions.reduce(
    (summary, prediction) => {
      summary[prediction.level] += 1;
      return summary;
    },
    { HIGH: 0, MEDIUM: 0, LOW: 0 } as Record<'HIGH' | 'MEDIUM' | 'LOW', number>
  );
  const totalPredictions = predictions.length;
  const riskWidth = (count: number) => `${totalPredictions ? (count / totalPredictions) * 100 : 0}%`;

  return (
    <Card className="prediction-panel" component="section" aria-labelledby="prediction-title">
      <CardContent>
      <Box className="prediction-heading">
        <div>
          <Typography className="dashboard-eyebrow"><Brain size={13} /> AI-assisted workforce insight</Typography>
          <Typography className="section-title" component="h2" id="prediction-title">Absenteeism Prediction</Typography>
          <Typography className="prediction-description">
            Risk levels generated from the prediction records currently available.
          </Typography>
        </div>
        <Box className="prediction-actions">
          {!loading && !error && <Chip className="prediction-total" label={`${predictions.length} records`} size="small" />}
          {onViewAll && <Button className="view-all-button" size="small" endIcon={<ArrowUpRight size={15} />} onClick={onViewAll}>View all</Button>}
        </Box>
      </Box>

      {error ? (
        <Box className="prediction-state prediction-error" role="alert"><span>{error}</span><Button onClick={onRetry}>Try again</Button></Box>
      ) : loading ? (
        <Box className="prediction-state"><Skeleton /><Skeleton /><Skeleton /></Box>
      ) : predictions.length === 0 ? (
        <Box className="prediction-state">
          No absenteeism predictions are available yet.
        </Box>
      ) : (
        <>
          <Box className="prediction-summary" aria-label="Prediction risk summary">
            <Box className="risk-summary risk-high"><Box className="risk-summary-label"><span>High risk</span><strong>{counts.HIGH}</strong></Box><LinearProgress variant="determinate" value={Number.parseFloat(riskWidth(counts.HIGH))} /></Box>
            <Box className="risk-summary risk-medium"><Box className="risk-summary-label"><span>Medium risk</span><strong>{counts.MEDIUM}</strong></Box><LinearProgress variant="determinate" value={Number.parseFloat(riskWidth(counts.MEDIUM))} /></Box>
            <Box className="risk-summary risk-low"><Box className="risk-summary-label"><span>Low risk</span><strong>{counts.LOW}</strong></Box><LinearProgress variant="determinate" value={Number.parseFloat(riskWidth(counts.LOW))} /></Box>
          </Box>

          <Box className="prediction-list">
            <div className="prediction-list-header">
              <span>Employee</span>
              <span>Risk score</span>
              <span>Level</span>
              <span>Prediction date</span>
            </div>
            {sortedPredictions.slice(0, 5).map((prediction) => {
              const user = userMap.get(prediction.userId);
              const department = user?.departementId
                ? departmentMap.get(user.departementId)
                : undefined;
              const employeeName = user
                ? `${user.firstName} ${user.lastName}`.trim()
                : `User #${prediction.userId}`;

              return (
                <Box className="prediction-row" key={prediction.id}>
                  <div className="prediction-employee">
                    <strong>{employeeName}</strong>
                    <span>{department || 'Department unavailable'}</span>
                  </div>
                  <strong className="prediction-score">{formatScore(prediction.score)}</strong>
                  <Chip className={`prediction-level level-${prediction.level.toLowerCase()}`} label={prediction.level} size="small" />
                  <span className="prediction-date">
                    <CalendarDays size={13} />
                    {formatDate(prediction.predictionDate)}
                    <small>{prediction.analysisPeriod}</small>
                  </span>
                </Box>
              );
            })}
          </Box>
          {predictions.length > 5 && (
            <p className="prediction-footnote">Showing the five highest risk records.</p>
          )}
        </>
      )}
      </CardContent>
    </Card>
  );
};

import React, { useCallback, useEffect, useState } from 'react';
import { RefreshCw } from 'lucide-react';
import { api } from '../../services/api';
import type {
  ContratResponseDto,
  EvaluationResponseDto,
  LeaveResponseDto,
  PresenceResponseDto,
  UserResponseDto,
} from '../../types/api';
import './AdminResourcePage.css';

type ResourceName = 'contracts' | 'leaves' | 'presences' | 'evaluations';
type ResourceRow = ContratResponseDto | LeaveResponseDto | PresenceResponseDto | EvaluationResponseDto;

interface AdminResourcePageProps {
  resource: ResourceName;
  title: string;
  subtitle: string;
}

const formatDate = (value?: string) => value ? new Date(value).toLocaleDateString() : '-';
const displayValue = (value?: string | number | boolean) => value === undefined || value === null ? '-' : String(value);

export const AdminResourcePage: React.FC<AdminResourcePageProps> = ({ resource, title, subtitle }) => {
  const [rows, setRows] = useState<ResourceRow[]>([]);
  const [users, setUsers] = useState<UserResponseDto[]>([]);
  const [loading, setLoading] = useState(true);
  const [errorMsg, setErrorMsg] = useState('');

  const loadData = useCallback(async () => {
    setLoading(true);
    setErrorMsg('');
    try {
      const [resourceResult, usersResult] = await Promise.allSettled([
        resource === 'contracts'
          ? api.getContrats()
          : resource === 'leaves'
            ? api.getLeaves()
            : resource === 'presences'
              ? api.getPresences()
              : api.getEvaluations(),
        api.getUsers(),
      ]);
      if (resourceResult.status === 'rejected') throw resourceResult.reason;
      setRows(resourceResult.value as ResourceRow[]);
      if (usersResult.status === 'fulfilled') setUsers(usersResult.value);
      else console.error('User lookup data unavailable:', usersResult.reason);
    } catch (error: any) {
      console.error(`Error loading ${resource}:`, error);
      const status = error.response?.status;
      setErrorMsg(status === 401 || status === 403
        ? `Access denied (${status}). This area is read-only and requires permission from the backend.`
        : error.message || `Unable to load ${title.toLowerCase()}.`);
    } finally {
      setLoading(false);
    }
  }, [resource, title]);

  useEffect(() => {
    loadData();
  }, [loadData]);

  const userName = (userId: number) => {
    const user = users.find((item) => item.id === userId);
    return user ? `${user.firstName} ${user.lastName}`.trim() : `User #${userId}`;
  };

  const renderRows = () => {
    if (resource === 'contracts') {
      return (rows as ContratResponseDto[]).map((contract) => (
        <tr key={contract.id}>
          <td>{userName(contract.userId)}</td>
          <td>{contract.type}</td>
          <td>{formatDate(contract.startDate)}</td>
          <td>{formatDate(contract.endDate)}</td>
          <td><span className="resource-status">{contract.status}</span></td>
        </tr>
      ));
    }
    if (resource === 'leaves') {
      return (rows as LeaveResponseDto[]).map((leave) => (
        <tr key={leave.id}>
          <td>{userName(leave.userId)}</td>
          <td>{leave.type}</td>
          <td>{formatDate(leave.startDate)}</td>
          <td>{formatDate(leave.endDate)}</td>
          <td><span className="resource-status">{leave.status}</span></td>
        </tr>
      ));
    }
    if (resource === 'presences') {
      return (rows as PresenceResponseDto[]).map((presence) => (
        <tr key={presence.id}>
          <td>{userName(presence.userId)}</td>
          <td>{formatDate(presence.date)}</td>
          <td>{displayValue(presence.checkInTime)}</td>
          <td>{displayValue(presence.checkOutTime)}</td>
          <td><span className="resource-status">{presence.status}</span></td>
        </tr>
      ));
    }
    return (rows as EvaluationResponseDto[]).map((evaluation) => {
      const scores = [evaluation.performanceScore, evaluation.competencyScore, evaluation.behaviorScore];
      const average = scores.every((score) => typeof score === 'number')
        ? (scores.reduce((total, score) => total + score, 0) / scores.length).toFixed(1)
        : '-';
      return (
        <tr key={evaluation.id}>
          <td>{userName(evaluation.evaluatedUserId)}</td>
          <td>{evaluation.period}</td>
          <td>{average}</td>
          <td>{formatDate(evaluation.evaluationDate)}</td>
          <td><span className="resource-status">{evaluation.confirmed ? 'CONFIRMED' : 'DRAFT'}</span></td>
        </tr>
      );
    });
  };

  const headers = {
    contracts: ['Employee', 'Type', 'Start date', 'End date', 'Status'],
    leaves: ['Employee', 'Type', 'Start date', 'End date', 'Status'],
    presences: ['Employee', 'Date', 'Check in', 'Check out', 'Status'],
    evaluations: ['Employee', 'Period', 'Average score', 'Evaluation date', 'State'],
  }[resource];

  return (
    <section className="resource-page">
      <div className="resource-page-header">
        <div>
          <p className="eyebrow">Read-only overview</p>
          <h1>{title}</h1>
          <p>{subtitle}</p>
        </div>
        <button className="resource-refresh" onClick={loadData} disabled={loading}>
          <RefreshCw size={15} style={{ marginRight: 6 }} />
          {loading ? 'Refreshing...' : 'Refresh'}
        </button>
      </div>
      {errorMsg && (
        <div className="resource-error" role="alert">
          <span>{errorMsg}</span>
          <button onClick={loadData}>Try again</button>
        </div>
      )}
      <div className="resource-card">
        <div className="resource-card-meta">
          <strong>{loading ? '...' : rows.length}</strong>
          <span>records available</span>
        </div>
        <div className="resource-table-wrap">
          <table className="resource-table">
            <thead>
              <tr>
                {headers.map((header) => (
                  <th key={header}>{header}</th>
                ))}
              </tr>
            </thead>
            <tbody>
              {loading ? (
                <tr>
                  <td colSpan={headers.length} className="resource-state">
                    Loading {title.toLowerCase()}...
                  </td>
                </tr>
              ) : rows.length === 0 ? (
                <tr>
                  <td colSpan={headers.length} className="resource-state">
                    No {title.toLowerCase()} records found.
                  </td>
                </tr>
              ) : (
                renderRows()
              )}
            </tbody>
          </table>
        </div>
      </div>
    </section>
  );
};

import React, { useEffect } from 'react';
import { authService } from '../services/authService';

interface ProtectedRouteProps {
  requiredRole?: string;
  onNavigate: (path: string) => void;
  children: React.ReactNode;
}

export const ProtectedRoute: React.FC<ProtectedRouteProps> = ({
  requiredRole = 'ADMIN',
  onNavigate,
  children,
}) => {
  const role = authService.getRole();

  const isAuthenticated = authService.isAuthenticated();
  const hasRequiredRole = !requiredRole || role === requiredRole;

  useEffect(() => {
    if (!isAuthenticated || !hasRequiredRole) {
      onNavigate('/login');
    }
  }, [isAuthenticated, hasRequiredRole, onNavigate]);

  if (!isAuthenticated || !hasRequiredRole) {
    return null;
  }

  return <>{children}</>;
};

interface LoginPageProps {
  onNavigate: (path: string) => void;
}

export const LoginPage: React.FC<LoginPageProps> = ({ onNavigate }) => {
  const [email, setEmail] = useState('');
  const [password, setPassword] = useState('');
  const [loading, setLoading] = useState(false);
  const [errorMsg, setErrorMsg] = useState('');

  const handleSubmit = async (e: React.FormEvent) => {
  e.preventDefault();
  setLoading(true);
  setErrorMsg('');

  try {
    const authenticatedUser = await authService.login({ email, password });
    if (authenticatedUser.role === 'ADMIN') {

    setErrorMsg(err.message || 'Login failed. Please check your credentials.');

    setLoading(false);
  }
};

  return (
    <div className="login-container">

      <div className="login-left">

          <Logo textColor="white" onClick={() => onNavigate('/')} />

import React, { useEffect, useState } from 'react';
import { Tooltip } from '@mui/material';
import { Pencil, Plus, Trash2, X } from 'lucide-react';
import { api } from '../../api/client';
import type { DepartementResponseDto } from '../../types/api';
import './DepartementsPage.css';

export const DepartementsPage: React.FC = () => {
  const [departements, setDepartements] = useState<DepartementResponseDto[]>([]);
  const [loading, setLoading] = useState<boolean>(true);
  const [errorMsg, setErrorMsg] = useState<string>('');

  const [isCreateOpen, setIsCreateOpen] = useState<boolean>(false);
  const [editingDept, setEditingDept] = useState<DepartementResponseDto | null>(null);

  const [name, setName] = useState('');
  const [description, setDescription] = useState('');

  const loadDepartements = async () => {
    setLoading(true);
    setErrorMsg('');
    try {
      const data = await api.getDepartements();
      setDepartements(data);
    } catch (err: any) {
      console.error('Error loading departements:', err);
      const status = err.response?.status;
      setErrorMsg(status === 401 || status === 403
        ? `Department access denied (${status}). Please sign in again or contact an administrator.`
        : err.message || 'Failed to load departments');
    } finally {
      setLoading(false);
    }
  };

  useEffect(() => {
    loadDepartements();
  }, []);

  const openCreateModal = () => {
    setName('');
    setDescription('');
    setIsCreateOpen(true);
  };

  const openEditModal = (dept: DepartementResponseDto) => {
    setEditingDept(dept);
    setName(dept.name || '');
    setDescription(dept.description || '');
  };

  const handleCreateSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    try {
      await api.createDepartement({ name, description });
      setIsCreateOpen(false);
      loadDepartements();
    } catch (err: any) {
      alert('Error creating department: ' + err.message);
    }
  };

  const handleEditSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!editingDept) return;
    try {
      await api.updateDepartement(editingDept.id, { name, description });
      setEditingDept(null);
      loadDepartements();
    } catch (err: any) {
      alert('Error updating department: ' + err.message);
    }
  };

  const handleDelete = async (id: number) => {
    if (!window.confirm('Are you sure you want to delete this department?')) return;
    try {
      await api.deleteDepartement(id);
      loadDepartements();
    } catch (err: any) {
      alert('Error deleting department: ' + err.message);
    }
  };

  return (
    <div className="page-container">
      <div className="page-header">
        <div>
          <h1 className="page-title">Department Management</h1>
          <p className="page-subtitle">Organize and maintain organizational departments</p>
        </div>
        <button className="primary-btn" onClick={openCreateModal}>
          <Plus size={16} /> Add Department
        </button>
      </div>

      {errorMsg && (
        <div className="page-error" role="alert">
          <span>{errorMsg}</span>
          <button onClick={loadDepartements}>Try again</button>
        </div>
      )}

      <div className="table-wrapper">
        <table className="data-table">
          <thead>
            <tr>
              <th>ID</th>
              <th>Department Name</th>
              <th>Description</th>
              <th>Actions</th>
            </tr>
          </thead>
          <tbody>
            {loading ? (
              <tr>
                <td colSpan={4} style={{ textAlign: 'center', padding: '24px' }}>
                  Loading departments...
                </td>

import type {
  ContratResponseDto,
  EvaluationResponseDto,
  LeaveResponseDto,
  PredictionResponseDto,
  PresenceResponseDto,
} from '../../types/api';

export interface DashboardMetrics {
  contractsToReview: number;
  leavesToday: LeaveResponseDto[];
  presentToday: number;
  absentToday: number;
  recordedAbsenceRate: number | null;
  overallEvaluationScore: number | null;
  highRiskPredictions: number;
  mediumRiskPredictions: number;
  lowRiskPredictions: number;
}

export const calculateDashboardMetrics = (
  contracts: ContratResponseDto[],
  leaves: LeaveResponseDto[],
  presences: PresenceResponseDto[],
  evaluations: EvaluationResponseDto[],
  predictions: PredictionResponseDto[],
  today: string
): DashboardMetrics => {
  const leavesToday = leaves.filter(
    (leave) => leave.status === 'APPROVED' && leave.startDate <= today && leave.endDate >= today
  );
  const todayPresences = presences.filter((presence) => presence.date === today);
  const measurablePresences = presences.filter((presence) => presence.status !== 'ON_LEAVE');
  const evaluationScores = evaluations.flatMap((evaluation) => [
    evaluation.performanceScore,
    evaluation.competencyScore,
    evaluation.behaviorScore,
  ]);

  return {
    contractsToReview: contracts.filter((contract) => ['EXPIRED', 'SUSPENDED'].includes(contract.status)).length,
    leavesToday,
    presentToday: todayPresences.filter((presence) => ['PRESENT', 'LATE'].includes(presence.status)).length,
    absentToday: todayPresences.filter((presence) => presence.status === 'ABSENT').length,
    recordedAbsenceRate: measurablePresences.length
      ? measurablePresences.filter((presence) => presence.status === 'ABSENT').length / measurablePresences.length * 100
      : null,
    overallEvaluationScore: evaluationScores.length
      ? evaluationScores.reduce((total, score) => total + score, 0) / evaluationScores.length
      : null,
    highRiskPredictions: predictions.filter((prediction) => prediction.level === 'HIGH').length,
    mediumRiskPredictions: predictions.filter((prediction) => prediction.level === 'MEDIUM').length,
    lowRiskPredictions: predictions.filter((prediction) => prediction.level === 'LOW').length,
  };
};

    if (open) setValues(initialValues);
  }, [open, initialValues]);

  if (!open) return null;

  const setValue = <K extends keyof UserFormValues>(key: K, value: UserFormValues[K]) => {
    setValues((current) => ({ ...current, [key]: value }));
  };

  const handleSubmit = (event: React.FormEvent<HTMLFormElement>) => {
    event.preventDefault();
    onSubmit(values);
  };

  const isCreate = mode === 'create';

  return (
    <div className="custom-modal-overlay" onClick={submitting ? undefined : onClose}>
      <div className="custom-modal-card" onClick={(e) => e.stopPropagation()}>
        <div className="custom-modal-header">
          <div>
            <div className="custom-modal-category">ADMINISTRATION</div>
            <h2 className="custom-modal-title">{isCreate ? 'Add user' : 'Edit user'}</h2>
          </div>
          <button
            type="button"
            className="custom-modal-close-btn"
            onClick={onClose}
            disabled={submitting}
            aria-label="Close modal"
          >
            <X size={18} />
          </button>
        </div>

        {error && <div className="custom-modal-error">{error}</div>}

        <form onSubmit={handleSubmit} className="custom-modal-form">
          <div className="custom-form-grid">
            {/* Row 1: First name & Last name */}
            <div className="custom-form-group">
              <label htmlFor="user-firstname">First name</label>
              <input
                id="user-firstname"
                type="text"
                value={values.firstName}
                onChange={(e) => setValue('firstName', e.target.value)}
                required
                placeholder="First name"
              />
            </div>

            <div className="custom-form-group">
              <label htmlFor="user-lastname">Last name</label>
              <input
                id="user-lastname"
                type="text"
                value={values.lastName}
                onChange={(e) => setValue('lastName', e.target.value)}
                required
                placeholder="Last name"
              />
            </div>

            {/* Row 2: Email & Phone */}
            <div className="custom-form-group">
              <label htmlFor="user-email">Email</label>
              <input
                id="user-email"
                type="email"
                value={values.email}
                onChange={(e) => setValue('email', e.target.value)}
                required
                disabled={!isCreate}
                placeholder="Email"
              />
            </div>

            <div className="custom-form-group">
              <label htmlFor="user-phone">Phone</label>
              <input
                id="user-phone"
                type="text"
                value={values.phoneNumber}
                onChange={(e) => setValue('phoneNumber', e.target.value)}
                required
                placeholder="Phone"
              />
            </div>

            {/* Row 3: Password & Hire date */}
            {isCreate ? (
              <div className="custom-form-group">
                <label htmlFor="user-password">Password</label>
                <input
                  id="user-password"
                  type="password"
                  value={values.password}
                  onChange={(e) => setValue('password', e.target.value)}
                  required
                  minLength={8}
                  placeholder="Password"
                />
              </div>
            ) : (
              <div className="custom-form-group">
                <label htmlFor="user-department">Department</label>
                <select
                  id="user-department"
                  value={values.departementId ?? ''}
                  onChange={(e) => setValue('departementId', e.target.value ? Number(e.target.value) : undefined)}
                >
                  <option value="" disabled>Select department</option>
                  {departments.map((dept) => (
                    <option key={dept.id} value={dept.id}>
                      {dept.name}
                    </option>
                  ))}

              </div>
            )}

                type="date"
                value={values.hireDate}
                onChange={(e) => setValue('hireDate', e.target.value)}
                required
              />
            </div>

            {/* Row 4: Status & Role */}
            <div className="custom-form-group">
              <label htmlFor="user-status">Status</label>
              <select
                id="user-status"
                value={values.status ?? 'ACTIVE'}
                onChange={(e) => setValue('status', e.target.value as UserStatus)}
                required
              >
                <option value="ACTIVE">Active</option>
                <option value="EN_LEAVE">On leave</option>
                <option value="SUSPENDED">Suspended</option>
                <option value="RESIGNED">Resigned</option>
                <option value="TERMINATED">Terminated</option>

            <div className="custom-form-group">

            </button>
          </div>

      </div>
    </div>
  );

  createDefaultUserFormValues,
  toCreatePayload,
  toUpdatePayload,
  toUserFormValues,
  UserFormDialog,
  type UserFormValues,
} from './components/UserFormDialog';
import './UsersPage.css';

export const UsersPage: React.FC = () => {
  const [users, setUsers] = useState<UserResponseDto[]>([]);
  const [departements, setDepartements] = useState<DepartementResponseDto[]>([]);
  const [positions, setPositions] = useState<PositionResponseDto[]>([]);
  const [roles, setRoles] = useState<RoleResponseDto[]>([]);
  const [loading, setLoading] = useState(true);
  const [errorMsg, setErrorMsg] = useState('');
  const [dialogMode, setDialogMode] = useState<'create' | 'edit' | null>(null);
  const [editingUser, setEditingUser] = useState<UserResponseDto | null>(null);
  const [dialogValues, setDialogValues] = useState<UserFormValues>(() => createDefaultUserFormValues([], [], []));
  const [dialogError, setDialogError] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [deletingId, setDeletingId] = useState<number | null>(null);
  const [deleteTarget, setDeleteTarget] = useState<UserResponseDto | null>(null);
  const [successMessage, setSuccessMessage] = useState('');

  const loadData = async () => {
    setLoading(true);
    setErrorMsg('');
    const [usersResult, departmentsResult, positionsResult, rolesResult] = await Promise.allSettled([
      api.getUsers(),
      api.getDepartements(),
      api.getPositions(),
      api.getRoles(),
    ]);

    if (usersResult.status === 'fulfilled') setUsers(usersResult.value);
    else {
      console.error('Error fetching users:', usersResult.reason);
      const status = usersResult.reason?.response?.status;
      setErrorMsg(status === 401 || status === 403
        ? `User directory access denied (${status}). Please sign in again or contact an administrator.`
        : usersResult.reason?.message || 'Failed to load user data');
    }
    if (departmentsResult.status === 'fulfilled') setDepartements(departmentsResult.value);
    if (positionsResult.status === 'fulfilled') setPositions(positionsResult.value);
    if (rolesResult.status === 'fulfilled') setRoles(rolesResult.value);
    if ([departmentsResult, positionsResult, rolesResult].some((result) => result.status === 'rejected') && usersResult.status === 'fulfilled') {
      console.error('Some user metadata requests failed:', { departmentsResult, positionsResult, rolesResult });
      setErrorMsg('Users loaded, but some department, position, or role details are unavailable.');
    }
    setLoading(false);
  };

  useEffect(() => { loadData(); }, []);

  const openCreateDialog = () => {
    setDialogError('');
    setEditingUser(null);
    setDialogValues(createDefaultUserFormValues(departements, positions, roles));
    setDialogMode('create');
  };

  const openEditDialog = (user: UserResponseDto) => {
    setDialogError('');
    setEditingUser(user);
    setDialogValues(toUserFormValues(user));
    setDialogMode('edit');
  };

  const closeDialog = () => {
    if (!submitting) {
      setDialogMode(null);
      setEditingUser(null);
      setDialogError('');
    }
  };

  const errorMessage = (error: unknown, fallback: string) => {
    if (typeof error === 'object' && error !== null && 'response' in error) {
      const response = (error as { response?: { status?: number; data?: { message?: string } } }).response;
      if (response?.data?.message) return response.data.message;
      if (response?.status === 400) return 'The request is invalid. Check the user details and try again.';
      if (response?.status === 401) return 'Your session is no longer authorized. Please sign in again.';
      if (response?.status === 403) return 'You are not authorized to perform this action.';
      if (response?.status === 404) return 'The user no longer exists.';
      if (response?.status === 409) return 'This user cannot be changed because it conflicts with existing data.';
      if (response?.status === 500) return 'The server could not complete this action. Existing HR records may reference this user.';
    }
    return error instanceof Error ? error.message : fallback;
  };

  const handleDialogSubmit = async (values: UserFormValues) => {
    const currentEmail = localStorage.getItem('email');
    const currentRole = localStorage.getItem('role');
    const selectedRole = roles.find((role) => role.id === values.roleId)?.name;
    if (
      dialogMode === 'edit' &&
      editingUser &&
      currentEmail &&
      editingUser.email.toLowerCase() === currentEmail.toLowerCase() &&
      currentRole === 'ADMIN' &&
      selectedRole !== 'ADMIN'
    ) {
      setDialogError('You cannot remove your own ADMIN access from this dialog.');
      return;
    }
    setSubmitting(true);
    setDialogError('');
    try {
      if (dialogMode === 'create') {
        await api.createUser(toCreatePayload(values));
        setSuccessMessage('User created successfully.');
      } else if (editingUser) {
        await api.updateUser(editingUser.id, toUpdatePayload(values));
        setSuccessMessage('User updated successfully.');
      }
      setDialogMode(null);
      setEditingUser(null);
      await loadData();
    } catch (error: unknown) {
      setDialogError(errorMessage(error, 'Unable to save user details.'));
    } finally {
      setSubmitting(false);
    }
  };

  const requestDelete = (user: UserResponseDto) => {
    const currentEmail = localStorage.getItem('email');
    if (currentEmail && user.email.toLowerCase() === currentEmail.toLowerCase()) {
      setErrorMsg('You cannot delete the currently authenticated administrator.');
      return;
    }
    setDeleteTarget(user);
  };

  const handleDelete = async () => {
    if (!deleteTarget) return;
    setDeletingId(deleteTarget.id);
    try {
      await api.deleteUser(deleteTarget.id);
      setSuccessMessage('User deleted successfully.');
      setDeleteTarget(null);
      await loadData();
    } catch (error: unknown) {
      setErrorMsg(errorMessage(error, 'Unable to delete user.'));
    } finally {
      setDeletingId(null);
    }
  };

  const getDeptName = (id?: number) => departements.find((department) => department.id === id)?.name || '-';
  const getRoleName = (id?: number) => roles.find((role) => role.id === id)?.name || '-';
  const getPositionTitle = (id?: number) => positions.find((position) => position.id === id)?.title || '-';

  return (
    <div className="page-container">
      <div className="page-header">
        <div>
          <h1 className="page-title">User Management</h1>
          <p className="page-subtitle">Manage accounts, roles, and workforce directory</p>
        </div>
        <button className="primary-btn" onClick={openCreateDialog}>+ Add user</button>
      </div>

      {errorMsg && (
        <div className="page-error" role="alert">
          <span>{errorMsg}</span>
          <button onClick={loadData}>Try again</button>
        </div>
      )}

      <div className="table-wrapper">
        <table className="data-table">
          <thead>
            <tr>
              <th>ID</th>
              <th>User</th>
              <th>Email</th>
              <th>Phone</th>
              <th>Department</th>
              <th>Role</th>
              <th>Position</th>
              <th>Status</th>
              <th>Actions</th>
            </tr>
          </thead>
          <tbody>
            {loading ? (
              <tr>
                <td colSpan={9} className="table-state">Loading users...</td>
              </tr>
            ) : users.length === 0 ? (
              <tr>
                <td colSpan={9} className="table-state">No users found.</td>
              </tr>
            ) : (
              users.map((user) => (
                <tr key={user.id}>
                  <td>#{user.id}</td>
                  <td className="primary-cell">{user.firstName} {user.lastName}</td>
                  <td>{user.email}</td>
                  <td>{user.phoneNumber || '-'}</td>
                  <td>{getDeptName(user.departementId)}</td>
                  <td>{getRoleName(user.roleId)}</td>
                  <td>{getPositionTitle(user.positionId)}</td>
                  <td>
                    <span
                      className={`status-tag ${
                        user.status === 'ACTIVE'
                          ? 'status-active'
                          : user.status === 'SUSPENDED'
                          ? 'status-suspended'
                          : user.status === 'EN_LEAVE'
                          ? 'status-leave'
                          : 'status-resigned'
                      }`}
                    >
                      {user.status || 'ACTIVE'}
                    </span>
                  </td>
                  <td>
                    <div className="action-btns">
                      <Tooltip title="Edit user">
                        <button
                          type="button"
                          className="action-icon-btn pencil-btn"
                          onClick={() => openEditDialog(user)}
                          aria-label={`Edit ${user.email}`}
                        >
                          <Pencil size={16} />
                        </button>
                      </Tooltip>
                      <Tooltip title="Delete user">
                        <button
                          type="button"
                          className="action-icon-btn trash-btn"
                          onClick={() => requestDelete(user)}
                          disabled={deletingId === user.id}
                          aria-label={`Delete ${user.email}`}
                        >
                          <Trash2 size={16} />
                        </button>
                      </Tooltip>
                    </div>
                  </td>
                </tr>
              ))
            )}
          </tbody>
        </table>
      </div>

      {dialogMode && (
        <UserFormDialog
          open
          mode={dialogMode}
          initialValues={dialogValues}
          departments={departements}
          positions={positions}
          roles={roles}
          users={users}
          submitting={submitting}
          error={dialogError}
          onClose={closeDialog}
          onSubmit={handleDialogSubmit}
        />
      )}

      <Dialog
        open={Boolean(deleteTarget)}
        onClose={deletingId === null ? () => setDeleteTarget(null) : undefined}
        maxWidth="xs"
        fullWidth
        slotProps={{
          paper: { style: { borderRadius: 16, padding: 8 } }
        }}
      >
        <DialogTitle style={{ fontWeight: 700, fontSize: '18px' }}>Delete user?</DialogTitle>
        <DialogContent>
          <Alert severity="warning" style={{ borderRadius: 10 }}>
            This will permanently remove {deleteTarget?.firstName} {deleteTarget?.lastName} ({deleteTarget?.email}). Existing HR records referencing this user may prevent deletion.
          </Alert>
        </DialogContent>
        <DialogActions style={{ padding: '8px 24px 16px 24px' }}>
          <Button
            onClick={() => setDeleteTarget(null)}
            disabled={deletingId !== null}
            style={{ color: '#475569', fontWeight: 600 }}
          >
            Cancel
          </Button>
          <Button
            color="error"
            variant="contained"
            onClick={handleDelete}
            disabled={deletingId !== null}
            startIcon={deletingId !== null ? <span className="button-spinner" /> : <Trash2 size={16} />}
            style={{ borderRadius: 8, fontWeight: 600, backgroundColor: '#DC2626' }}
          >
            {deletingId !== null ? 'Deleting...' : 'Delete user'}
          </Button>
        </DialogActions>
      </Dialog>

      <Snackbar
        open={Boolean(successMessage)}
        autoHideDuration={3500}
        onClose={() => setSuccessMessage('')}
      >
        <Alert severity="success" onClose={() => setSuccessMessage('')}>
          {successMessage}
        </Alert>
      </Snackbar>
    </div>
  );
};

import React, { useCallback, useEffect, useMemo, useRef, useState } from 'react';
import { Alert, Snackbar } from '@mui/material';
import { Check, ChevronDown, Shield } from 'lucide-react';
import { api } from '../../api/client';
import { useAdminRefresh } from '../../components/admin/AdminRefreshContext';
import type { RoleResponseDto, UserResponseDto } from '../../types/api';
import './RolesPage.css';

const userLabel = (user: UserResponseDto) =>
  `${user.firstName} ${user.lastName} (${user.email})`;

export const RolesPage: React.FC = () => {
  const { register } = useAdminRefresh();
  const [users, setUsers] = useState<UserResponseDto[]>([]);
  const [roles, setRoles] = useState<RoleResponseDto[]>([]);
  const [loading, setLoading] = useState(true);
  const [errorMsg, setErrorMsg] = useState('');
  const [selectedUserId, setSelectedUserId] = useState<number | ''>('');
  const [selectedRoleId, setSelectedRoleId] = useState<number | ''>('');
  const [openMenu, setOpenMenu] = useState<'user' | 'role' | null>(null);
  const [submitting, setSubmitting] = useState(false);
  const [successMessage, setSuccessMessage] = useState('');
  const formRef = useRef<HTMLFormElement>(null);

  const loadData = useCallback(async () => {
    setLoading(true);
    setErrorMsg('');
    const [usersResult, rolesResult] = await Promise.allSettled([
      api.getUsers(),
      api.getRoles(),
    ]);

    if (usersResult.status === 'fulfilled') {
      setUsers(usersResult.value);
    } else {
      console.error('Error fetching users for roles page:', usersResult.reason);
      const status = usersResult.reason?.response?.status;
      setErrorMsg(
        status === 401 || status === 403
          ? `User directory access denied (${status}). Please sign in again or contact an administrator.`
          : usersResult.reason?.message || 'Failed to load users'
      );
    }

    if (rolesResult.status === 'fulfilled') {
      setRoles(rolesResult.value);
    } else if (usersResult.status === 'fulfilled') {
      console.error('Error fetching roles:', rolesResult.reason);
      setErrorMsg('Users loaded, but role options are unavailable.');
    }

    setLoading(false);
  }, []);

  useEffect(() => {
    loadData();
  }, [loadData]);

  useEffect(() => register(loadData, loading), [register, loadData, loading]);

  useEffect(() => {
    const handlePointerDown = (event: MouseEvent) => {
      if (formRef.current && !formRef.current.contains(event.target as Node)) {
        setOpenMenu(null);
      }
    };
    document.addEventListener('mousedown', handlePointerDown);
    return () => document.removeEventListener('mousedown', handlePointerDown);
  }, []);

  const selectedUser = useMemo(
    () => users.find((user) => user.id === selectedUserId),
    [users, selectedUserId]
  );
  const selectedRole = useMemo(
    () => roles.find((role) => role.id === selectedRoleId),
    [roles, selectedRoleId]
  );

  const selectUser = (userId: number) => {
    const user = users.find((entry) => entry.id === userId);
    setSelectedUserId(userId);
    setSelectedRoleId(user?.roleId ?? '');
    setOpenMenu(null);
    setErrorMsg('');
  };

  const selectRole = (roleId: number | '') => {
    setSelectedRoleId(roleId);
    setOpenMenu(null);
  };

  const errorMessage = (error: unknown, fallback: string) => {
    if (typeof error === 'object' && error !== null && 'response' in error) {
      const response = (error as { response?: { status?: number; data?: { message?: string } } }).response;
      if (response?.data?.message) return response.data.message;
      if (response?.status === 400) return 'The request is invalid. Check the selected user and role.';
      if (response?.status === 401) return 'Your session is no longer authorized. Please sign in again.';
      if (response?.status === 403) return 'You are not authorized to change this role.';
      if (response?.status === 404) return 'The user or role no longer exists.';
      if (response?.status === 409) return 'This role cannot be changed because it conflicts with existing data.';
    }
    return error instanceof Error ? error.message : fallback;
  };

  const handleSave = async (event: React.FormEvent) => {
    event.preventDefault();
    if (!selectedUser || selectedRoleId === '') {
      setErrorMsg('Select a user and a role before saving.');
      return;
    }

    const currentEmail = localStorage.getItem('email');
    const currentRole = localStorage.getItem('role');
    const nextRoleName = roles.find((role) => role.id === selectedRoleId)?.name;
    if (
      currentEmail &&
      selectedUser.email.toLowerCase() === currentEmail.toLowerCase() &&
      currentRole === 'ADMIN' &&
      nextRoleName !== 'ADMIN'
    ) {
      setErrorMsg('You cannot remove your own ADMIN access from this page.');
      return;
    }

    setSubmitting(true);
    setErrorMsg('');
    try {
      await api.updateUser(selectedUser.id, {
        firstName: selectedUser.firstName,
        lastName: selectedUser.lastName,
        phoneNumber: selectedUser.phoneNumber || '',
        status: selectedUser.status || 'ACTIVE',
        departementId: selectedUser.departementId,
        roleId: selectedRoleId,
        positionId: selectedUser.positionId,
        managerId: selectedUser.managerId,
      });
      setSuccessMessage(`Role updated for ${userLabel(selectedUser)}.`);
      await loadData();
    } catch (error: unknown) {
      setErrorMsg(errorMessage(error, 'Unable to save the assigned role.'));
    } finally {
      setSubmitting(false);
    }
  };

  const saveDisabled =
    loading ||
    submitting ||
    !selectedUser ||
    selectedRoleId === '' ||
    selectedUser.roleId === selectedRoleId;

  return (
    <div className="roles-page">
      <div className="roles-page-header">
        <div>
          <p className="roles-eyebrow">ADMINISTRATION</p>
          <h1 className="roles-title">Role management</h1>
          <p className="roles-subtitle">Change the role assigned to an existing user.</p>
        </div>
        <div className="roles-header-icon" aria-hidden="true">
          <Shield size={28} strokeWidth={1.75} />
        </div>
      </div>

      {errorMsg && (
        <div className="page-error" role="alert">
          <span>{errorMsg}</span>
          <button type="button" onClick={loadData}>Try again</button>
        </div>
      )}

      <form className="roles-card" onSubmit={handleSave} ref={formRef}>
        <div className="roles-field">
          <label htmlFor="roles-user-trigger">User</label>
          <div className="roles-select">
            <button
              id="roles-user-trigger"
              type="button"
              className="roles-select-trigger"
              aria-haspopup="listbox"
              aria-expanded={openMenu === 'user'}
              disabled={loading || users.length === 0}
              onClick={() => setOpenMenu((current) => (current === 'user' ? null : 'user'))}
            >
              <span className={selectedUser ? '' : 'roles-placeholder'}>
                {loading ? 'Loading users...' : selectedUser ? userLabel(selectedUser) : 'Select a user'}
              </span>
              <ChevronDown size={16} />
            </button>
            {openMenu === 'user' && (
              <ul className="roles-select-menu" role="listbox" aria-labelledby="roles-user-trigger">
                {users.length === 0 ? (
                  <li className="roles-select-empty">No users found.</li>
                ) : (
                  users.map((user) => (
                    <li key={user.id}>
                      <button
                        type="button"
                        role="option"
                        aria-selected={user.id === selectedUserId}
                        className={user.id === selectedUserId ? 'is-selected' : ''}
                        onClick={() => selectUser(user.id)}
                      >
                        {userLabel(user)}
                      </button>
                    </li>
                  ))
                )}
              </ul>
            )}
          </div>
        </div>

        <div className="roles-field">
          <label htmlFor="roles-role-trigger">Role</label>
          <div className="roles-select">
            <button
              id="roles-role-trigger"
              type="button"
              className="roles-select-trigger"
              aria-haspopup="listbox"
              aria-expanded={openMenu === 'role'}
              disabled={loading || !selectedUser || roles.length === 0}
              onClick={() => setOpenMenu((current) => (current === 'role' ? null : 'role'))}
            >
              <span className={selectedRole ? '' : 'roles-placeholder'}>
                {selectedRole ? selectedRole.name : 'Select a role'}
              </span>
              <ChevronDown size={16} />
            </button>
            {openMenu === 'role' && (
              <ul className="roles-select-menu" role="listbox" aria-labelledby="roles-role-trigger">
                <li>
                  <button
                    type="button"
                    role="option"
                    aria-selected={selectedRoleId === ''}
                    className={selectedRoleId === '' ? 'is-selected' : ''}
                    onClick={() => selectRole('')}
                  >
                    Select a role
                  </button>
                </li>
                {roles.map((role) => (
                  <li key={role.id}>
                    <button
                      type="button"
                      role="option"
                      aria-selected={role.id === selectedRoleId}
                      className={role.id === selectedRoleId ? 'is-selected' : ''}
                      onClick={() => selectRole(role.id)}
                    >
                      {role.name}
                    </button>
                  </li>
                ))}
              </ul>
            )}
          </div>
        </div>

        <button type="submit" className="roles-save-btn" disabled={saveDisabled}>
          <Check size={16} strokeWidth={2.5} />
          {submitting ? 'Saving...' : 'Save role'}
        </button>
      </form>
      {!loading && users.length === 0 && !errorMsg && (
        <p className="roles-empty">No users are available to assign a role.</p>
      )}

      <Snackbar
        open={Boolean(successMessage)}
        autoHideDuration={3500}
        onClose={() => setSuccessMessage('')}
      >
        <Alert severity="success" onClose={() => setSuccessMessage('')}>
          {successMessage}
        </Alert>
      </Snackbar>
    </div>
  );
};

  LeaveResponseDto,
  PredictionResponseDto,
  PresenceResponseDto,
  PositionResponseDto,

import './AdminDashboard.css';

interface AdminDashboardProps {
  onNavigate: (path: string) => void;
}

export const AdminDashboard: React.FC<AdminDashboardProps> = ({ onNavigate }) => {
  const { register } = useAdminRefresh();
  const [users, setUsers] = useState<UserResponseDto[]>([]);
  const [departements, setDepartements] = useState<DepartementResponseDto[]>([]);
  const [positions, setPositions] = useState<PositionResponseDto[]>([]);
  const [contracts, setContracts] = useState<ContratResponseDto[]>([]);
  const [leaves, setLeaves] = useState<LeaveResponseDto[]>([]);
  const [presences, setPresences] = useState<PresenceResponseDto[]>([]);
  const [evaluations, setEvaluations] = useState<EvaluationResponseDto[]>([]);
  const [predictions, setPredictions] = useState<PredictionResponseDto[]>([]);
  const [loading, setLoading] = useState(true);
  const [predictionLoading, setPredictionLoading] = useState(true);
  const [errorMsg, setErrorMsg] = useState('');
  const [predictionError, setPredictionError] = useState('');

  const loadOverview = useCallback(async () => {
    setLoading(true);
    setPredictionLoading(true);
    setErrorMsg('');
    setPredictionError('');

    const results = await Promise.allSettled([
      api.getUsers(), api.getDepartements(), api.getPredictions(), api.getPositions(),
      api.getContrats(), api.getLeaves(), api.getPresences(), api.getEvaluations(),
    ]);
    const [usersResult, departmentsResult, predictionsResult, positionsResult, contractsResult, leavesResult, presencesResult, evaluationsResult] = results;

    if (usersResult.status === 'fulfilled') setUsers(usersResult.value);
    if (departmentsResult.status === 'fulfilled') setDepartements(departmentsResult.value);
    if (positionsResult.status === 'fulfilled') setPositions(positionsResult.value);
    if (contractsResult.status === 'fulfilled') setContracts(contractsResult.value);
    if (leavesResult.status === 'fulfilled') setLeaves(leavesResult.value);
    if (presencesResult.status === 'fulfilled') setPresences(presencesResult.value);
    if (evaluationsResult.status === 'fulfilled') setEvaluations(evaluationsResult.value);
    if (predictionsResult.status === 'fulfilled') setPredictions(predictionsResult.value);

    if (predictionsResult.status === 'rejected') {
      console.error('Error loading absenteeism predictions:', predictionsResult.reason);
      const status = predictionsResult.reason?.response?.status;
      setPredictionError(status === 401 || status === 403 ? `Prediction access denied (${status}).` : predictionsResult.reason?.message || 'Prediction data could not be loaded.');
    }

    const failures = results.filter((result) => result.status === 'rejected');
    if (failures.length > 0) {
      console.error('Error loading dashboard overview:', failures);
      setErrorMsg('Some overview data could not be loaded. Available sections remain visible.');
    }
    setLoading(false);
    setPredictionLoading(false);
  }, []);

  useEffect(() => { loadOverview(); }, [loadOverview]);
  useEffect(() => register(loadOverview, loading), [loadOverview, loading, register]);

  const activeEmployees = users.filter((user) => user.status === 'ACTIVE').length;
  const today = new Date().toISOString().slice(0, 10);
  const metrics = calculateDashboardMetrics(contracts, leaves, presences, evaluations, predictions, today);

  return (
    <Box className="dashboard-container">
      {errorMsg && <Alert severity="warning" action={<Button color="inherit" size="small" onClick={loadOverview}>Retry</Button>}>{errorMsg}</Alert>}

      <Box className="dashboard-kpi-row">
        <WorkforceActions contractsToReview={metrics.contractsToReview} leavesToday={metrics.leavesToday.length} onNavigate={onNavigate} />
        <EvaluationScoreCard loading={loading} score={metrics.overallEvaluationScore} />
      </Box>
      <AttendanceOverview loading={loading} presentToday={metrics.presentToday} absentToday={metrics.absentToday} leaveToday={metrics.leavesToday.length} recordedAbsenceRate={metrics.recordedAbsenceRate} />
      <Box className="prediction-feature">
        <AbsenteeismPredictionPanel predictions={predictions} users={users} departements={departements} loading={predictionLoading} error={predictionError} onRetry={loadOverview} onViewAll={() => onNavigate('/admin/predictions')} />
      </Box>
      <DirectorySnapshot users={users} departments={departements} loading={loading} />
      <Box className="dashboard-source-note">Positions: {loading ? '...' : positions.length} · High-risk predictions: {predictionLoading ? '...' : metrics.highRiskPredictions} · Medium: {predictionLoading ? '...' : metrics.mediumRiskPredictions} · Low: {predictionLoading ? '...' : metrics.lowRiskPredictions}</Box>
    </Box>
  );
};

import type {
  ContratResponseDto,
  LeaveResponseDto,
  PredictionResponseDto,
  UserResponseDto,
} from '../../types/api';

export interface RhMetrics {
  totalEmployees: number;
  totalContracts: number;
  pendingLeaves: number;
  approvedLeaves: number;
  rejectedLeaves: number;
  highRiskCount: number;
  mediumRiskCount: number;
  lowRiskCount: number;
}

export const calculateRhMetrics = (
  users: UserResponseDto[],
  contracts: ContratResponseDto[],
  leaves: LeaveResponseDto[],
  predictions: PredictionResponseDto[]
): RhMetrics => ({
  totalEmployees: users.length,
  totalContracts: contracts.length,

  highRiskCount: predictions.filter((prediction) => prediction.level === 'HIGH').length,
  mediumRiskCount: predictions.filter((prediction) => prediction.level === 'MEDIUM').length,
  lowRiskCount: predictions.filter((prediction) => prediction.level === 'LOW').length,
});

import React, { useEffect, useState } from 'react';
import { api } from '../../api/client';
import type { DepartementResponseDto, UserResponseDto } from '../../types/api';
import './EmployeesPage.css';

export const EmployeesPage: React.FC = () => {
  const [users, setUsers] = useState<UserResponseDto[]>([]);
  const [departements, setDepartements] = useState<DepartementResponseDto[]>([]);
  const [loading, setLoading] = useState(true);
  const [errorMsg, setErrorMsg] = useState('');
  const [savingId, setSavingId] = useState<number | null>(null);

  const loadData = async () => {
    setLoading(true);
    setErrorMsg('');
    const [usersResult, deptResult] = await Promise.allSettled([
      api.getUsers(),
      api.getDepartements(),
    ]);
    if (usersResult.status === 'fulfilled') setUsers(usersResult.value);
    else {
      const status = (usersResult.reason as any)?.response?.status;
      setErrorMsg(status === 401 || status === 403 ? `Accès refusé (${status}).` : 'Impossible de charger les employés.');
    }
    if (deptResult.status === 'fulfilled') setDepartements(deptResult.value);
    setLoading(false);
  };

  useEffect(() => { loadData(); }, []);

  const getDeptName = (id?: number) => departements.find((d) => d.id === id)?.name || '-';

  const handleAssign = async (user: UserResponseDto, departementId: number | undefined) => {
    setSavingId(user.id);
    try {
      await api.updateUser(user.id, {
        firstName: user.firstName,
        lastName: user.lastName,
        phoneNumber: user.phoneNumber || '',
        status: user.status || 'ACTIVE',
        departementId,
        roleId: user.roleId,
        positionId: user.positionId,
        managerId: user.managerId,
      });
      await loadData();
    } catch (err: any) {
      alert('Erreur : ' + (err.message || 'impossible de mettre à jour le département.'));
    } finally {
      setSavingId(null);
    }
  };

  return (
    <div className="page-container">
      <div className="page-header">
        <div>
          <p className="page-eyebrow">People Operations</p>
          <h1 className="page-title">Employees</h1>
          <p className="page-subtitle">Consultez les employés et gérez leur affectation aux départements.</p>
        </div>
      </div>

      {errorMsg && <div className="page-error" role="alert"><span>{errorMsg}</span><button onClick={loadData}>Réessayer</button></div>}

      <div className="table-wrapper">
        <table className="data-table">
          <thead>
            <tr><th>Employé</th><th>Email</th><th>Département</th><th>Assigner à</th><th>Action</th></tr>
          </thead>
          <tbody>
            {loading ? (
              <tr><td colSpan={5} className="table-state">Chargement...</td></tr>
            ) : users.length === 0 ? (
              <tr><td colSpan={5} className="table-state">Aucun employé trouvé.</td></tr>
            ) : (
              users.map((user) => (
                <tr key={user.id}>
                  <td>{user.firstName} {user.lastName}</td>
                  <td>{user.email}</td>
                  <td>{getDeptName(user.departementId)}</td>
                  <td>
                    <select
                      className="dept-select"
                      value={user.departementId ?? ''}
                      disabled={savingId === user.id}
                      onChange={(e) => handleAssign(user, e.target.value ? Number(e.target.value) : undefined)}
                    >
                      <option value="">Aucun</option>
                      {departements.map((d) => (
                        <option key={d.id} value={d.id}>{d.name}</option>
                      ))}
                    </select>
                  </td>
                  <td>
                    <button
                      className="remove-dept-btn"
                      disabled={!user.departementId || savingId === user.id}
                      onClick={() => handleAssign(user, undefined)}
                    >
                      Retirer
                    </button>
                  </td>
                </tr>
              ))
            )}
          </tbody>
        </table>
      </div>
    </div>
  );
};

import { api } from './api';
import type { AuthResponseDto } from '../types/api';

export const authService = {
  async login(credentials: { email: string; password: string }): Promise<AuthResponseDto> {
    try {
      const response = await api.login(credentials);

      if (
        response.status === 200 &&
        response.data &&
        typeof response.data.token === 'string' &&
        response.data.token.trim() &&
        typeof response.data.email === 'string' &&
        response.data.email.trim() &&
        typeof response.data.role === 'string' &&
        response.data.role.trim()
      ) {
        localStorage.setItem('token', response.data.token);
        localStorage.setItem('email', response.data.email);
        localStorage.setItem('role', response.data.role);
        return response.data;
      } else {
        authService.logout();
        throw new Error('Login failed: Invalid server response.');
      }
    } catch (err: any) {
      authService.logout();
      const message = err.response?.data?.message || err.message || 'Invalid credentials';
      throw new Error(message);
    }
  },

  logout() {
    localStorage.removeItem('token');
    localStorage.removeItem('email');
    localStorage.removeItem('role');
  },

  getToken(): string | null {
    return localStorage.getItem('token');
  },

  getRole(): string | null {
    return localStorage.getItem('role');
  },

  isAuthenticated(): boolean {
    const token = authService.getToken();
    if (!token) return false;

    try {
      const payload = token.split('.')[1];
      if (!payload) return false;
      const normalizedPayload = payload.replace(/-/g, '+').replace(/_/g, '/');
      const paddedPayload = normalizedPayload.padEnd(Math.ceil(normalizedPayload.length / 4) * 4, '=');
      const claims = JSON.parse(atob(paddedPayload));
      return typeof claims.exp === 'number' && claims.exp * 1000 > Date.now();
    } catch {
      return false;
    }
  },
};

import React, { createContext, useContext, useState } from 'react';

interface RefreshContextType {
  refreshKey: number;
  triggerRefresh: () => void;
}

const RefreshContext = createContext<RefreshContextType | undefined>(undefined);

export const RefreshProvider: React.FC<{ children: React.ReactNode }> = ({ children }) => {
  const [refreshKey, setRefreshKey] = useState(0);

  const triggerRefresh = () => {
    setRefreshKey(prev => prev + 1);
  };

  return (
    <RefreshContext.Provider value={{ refreshKey, triggerRefresh }}>
      {children}
    </RefreshContext.Provider>
  );
};

export const useRefresh = (): RefreshContextType => {
  const context = useContext(RefreshContext);
  if (!context) {
    throw new Error('useRefresh must be used inside RefreshProvider');
  }
  return context;
};

import { useState, useEffect } from 'react';
import { LandingPage } from './pages/LandingPage';
import { LoginPage } from './pages/LoginPage';
import { ProtectedRoute } from './components/ProtectedRoute';

import { AdminLayout } from './components/admin/AdminLayout';
import { AdminDashboard } from './pages/admin/AdminDashboard';
import { UsersPage } from './pages/admin/UsersPage';
import { RolesPage } from './pages/admin/RolesPage';
import { DepartementsPage } from './pages/admin/DepartementsPage';
import { PositionsPage } from './pages/admin/PositionsPage';
import { PredictionsPage } from './pages/admin/PredictionsPage';
import { AdminResourcePage } from './components/admin/AdminResourcePage';

import { RhLayout } from './components/rh/RhLayout';
import { RhDashboard } from './pages/rh/RhDashboard';

export function App() {
  const [currentPath, setCurrentPath] = useState<string>(
    window.location.pathname
  );

  useEffect(() => {
    const handlePopState = () => {
      setCurrentPath(window.location.pathname);
    };

    window.addEventListener('popstate', handlePopState);

    return () => window.removeEventListener('popstate', handlePopState);
  }, []);

  const navigate = (path: string) => {
    window.history.pushState({}, '', path);
    setCurrentPath(path);
  };

  // Login
  if (currentPath === '/login') {
    return <LoginPage onNavigate={navigate} />;
  }

  // Admin routes
  if (currentPath.startsWith('/admin')) {
    return (
      <ProtectedRoute requiredRole="ADMIN" onNavigate={navigate}>
        <AdminLayout currentPath={currentPath} onNavigate={navigate}>
          
          {currentPath === '/admin/users' && <UsersPage />}

          {currentPath === '/admin/roles' && <RolesPage />}

          {currentPath === '/admin/departments' && <DepartementsPage />}

          {currentPath === '/admin/positions' && <PositionsPage />}

          {currentPath === '/admin/contracts' && (
            <AdminResourcePage
              resource="contracts"
              title="Contracts"
              subtitle="Consult workforce contract records and current statuses."
            />
          )}

          {currentPath === '/admin/leaves' && (
            <AdminResourcePage
              resource="leaves"
              title="Leaves"
              subtitle="Consult leave requests and approved absence periods."
            />
          )}

          {currentPath === '/admin/presences' && (
            <AdminResourcePage
              resource="presences"
              title="Présences"
              subtitle="Consult attendance and presence records across the workforce."
            />
          )}

          {currentPath === '/admin/evaluations' && (
            <AdminResourcePage
              resource="evaluations"
              title="Évaluations"
              subtitle="Consult performance, competency, and behavior evaluations."
            />
          )}

          {currentPath === '/admin/predictions' && <PredictionsPage />}

          {(currentPath === '/admin/dashboard' ||
            currentPath === '/admin') && (
            <AdminDashboard onNavigate={navigate} />
          )}

        </AdminLayout>
      </ProtectedRoute>
    );
  }

  // RH Manager routes
  if (currentPath.startsWith('/rh')) {
    return (
      <ProtectedRoute
        requiredRole="RH_MANAGER"
        onNavigate={navigate}
      >
        <RhLayout
          currentPath={currentPath}
          onNavigate={navigate}
        >
          {(currentPath === '/rh/dashboard' ||
            currentPath === '/rh') && (
            <RhDashboard />
          )}
        </RhLayout>
      </ProtectedRoute>
    );
  }

  // Default route
  return <LandingPage onNavigate={navigate} />;
}

export default App;
